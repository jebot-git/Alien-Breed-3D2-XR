cmake_minimum_required(VERSION 3.20)
project(AlienBreed3D2XR VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS native/src/core/*.cpp)
add_library(ab3d2-original STATIC ${CORE_SOURCES})
target_include_directories(ab3d2-original PUBLIC native/include native/src/core)
target_compile_options(ab3d2-original PRIVATE -fwrapv -fno-strict-aliasing -Wno-narrowing)
add_executable(ab3d2-core-test native/tests/core-main.cpp)
target_link_libraries(ab3d2-core-test PRIVATE ab3d2-original)
enable_testing()
find_package(nlohmann_json REQUIRED)
target_link_libraries(ab3d2-core-test PRIVATE nlohmann_json::nlohmann_json)
add_executable(ab3d2-arithmetic-test native/tests/arithmetic.cpp)
target_link_libraries(ab3d2-arithmetic-test PRIVATE ab3d2-original)
add_test(NAME NativeArithmetic COMMAND ab3d2-arithmetic-test)
set(AB3D2_REFERENCE_FIXTURES "${CMAKE_SOURCE_DIR}/artifacts/native-oracle" CACHE PATH "Local original reference states; optional development fixtures")
foreach(level RANGE 0 15)
  if(EXISTS "${AB3D2_REFERENCE_FIXTURES}/level-${level}.ram" AND EXISTS "${AB3D2_REFERENCE_FIXTURES}/frames-${level}.json")
    add_test(NAME NativeOriginalLevel${level} COMMAND ab3d2-core-test "${AB3D2_REFERENCE_FIXTURES}/level-${level}.ram" "${AB3D2_REFERENCE_FIXTURES}/frames-${level}.json")
    set_tests_properties(NativeOriginalLevel${level} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  endif()
endforeach()
find_package(ZLIB REQUIRED)
add_library(ab3d2-runtime STATIC native/src/runtime.cpp)
target_include_directories(ab3d2-runtime PUBLIC native/include)
target_link_libraries(ab3d2-runtime PUBLIC ab3d2-original nlohmann_json::nlohmann_json ZLIB::ZLIB)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(OPENXR REQUIRED IMPORTED_TARGET openxr)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(PNG REQUIRED)
find_package(X11 REQUIRED)
find_package(Threads REQUIRED)
target_link_libraries(ab3d2-runtime PUBLIC Threads::Threads)
add_library(ab3d2-presentation STATIC native/render/gpu.cpp native/render/render.cpp)
target_include_directories(ab3d2-presentation PUBLIC native/include)
target_link_libraries(ab3d2-presentation PUBLIC ab3d2-original OpenGL::GL GLEW::GLEW PNG::PNG nlohmann_json::nlohmann_json)
add_library(ab3d2-platform STATIC native/platform/xr.cpp native/platform/domed.cpp)
target_include_directories(ab3d2-platform PUBLIC native/include native/platform)
target_compile_definitions(ab3d2-platform PRIVATE TEMPEST_OPENXR=1)
target_link_libraries(ab3d2-platform PUBLIC PkgConfig::SDL2 PkgConfig::OPENXR PkgConfig::CURL OpenGL::GL OpenGL::EGL GLEW::GLEW X11::X11)
target_sources(ab3d2-original PRIVATE native/src/controls.cpp)
target_sources(ab3d2-platform PRIVATE native/platform/audio.cpp)
target_link_libraries(ab3d2-platform PUBLIC ab3d2-original)
add_executable(alien-breed3d2-xr native/src/main.cpp)
target_link_libraries(alien-breed3d2-xr PRIVATE ab3d2-runtime ab3d2-platform ab3d2-presentation)
add_executable(ab3d2-runtime-test native/tests/runtime.cpp)
target_link_libraries(ab3d2-runtime-test PRIVATE ab3d2-runtime)
if(EXISTS "${CMAKE_SOURCE_DIR}/assets/native/manifest.json")
 add_test(NAME NativeRuntime COMMAND ab3d2-runtime-test "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/runtime-test")
endif()
add_executable(ab3d2-controls-test native/tests/controls.cpp)
target_link_libraries(ab3d2-controls-test PRIVATE ab3d2-runtime)
if(EXISTS "${CMAKE_SOURCE_DIR}/assets/native/manifest.json")
 add_test(NAME NativeTrackedControls COMMAND ab3d2-controls-test "${CMAKE_SOURCE_DIR}/assets")
endif()
add_executable(ab3d2-audio-test native/platform/tests/audio-check.cpp)
target_link_libraries(ab3d2-audio-test PRIVATE ab3d2-original PkgConfig::SDL2)
target_compile_options(ab3d2-audio-test PRIVATE -UNDEBUG)
add_test(NAME NativePaulaAudio COMMAND ab3d2-audio-test)
option(AB3D2_BUILD_GPU_TESTS "Build isolated real-OpenGL renderer checks" OFF)
if(AB3D2_BUILD_GPU_TESTS)
 foreach(kind renderer effects)
  add_executable(ab3d2-${kind}-test native/render/tests/${kind}-check.cpp)
  target_include_directories(ab3d2-${kind}-test PRIVATE native/render)
  target_link_libraries(ab3d2-${kind}-test PRIVATE ab3d2-presentation PkgConfig::SDL2)
 endforeach()
 if(EXISTS "${CMAKE_SOURCE_DIR}/assets/native/manifest.json")
  add_test(NAME NativeGPUScenes COMMAND ab3d2-renderer-test "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_SOURCE_DIR}/assets/native" "${CMAKE_BINARY_DIR}/gpu-captures")
  add_test(NAME NativeGPUEffects COMMAND ab3d2-effects-test "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_SOURCE_DIR}/assets/native")
  set_tests_properties(NativeGPUScenes NativeGPUEffects PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen" RUN_SERIAL TRUE)
 endif()
endif()
include(GNUInstallDirs)
configure_file(native/packaging/launcher.in alien-breed3d2-xr-launcher @ONLY)
configure_file(native/packaging/alien-breed3d2-xr.desktop.in alien-breed3d2-xr.desktop @ONLY)
install(TARGETS alien-breed3d2-xr RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(PROGRAMS "${CMAKE_BINARY_DIR}/alien-breed3d2-xr-launcher" DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME alien-breed3d2-xr)
install(FILES "${CMAKE_BINARY_DIR}/alien-breed3d2-xr.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
# Original game data are local and never downloaded or generated by the build.
if(EXISTS "${CMAKE_SOURCE_DIR}/assets/native/manifest.json")
 install(DIRECTORY assets/ DESTINATION ${CMAKE_INSTALL_DATADIR}/alien-breed3d2-xr)
endif()
install(FILES native/README.md native/SOURCE_AUDIT.txt native/SOURCE_NOTICES.txt native/render/README.txt DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(AB3D2_BUILD_GPU_TESTS)
 add_library(ab3d2-mock-openxr SHARED native/tests/mock-openxr.cpp)
 target_link_libraries(ab3d2-mock-openxr PRIVATE OpenGL::GL OpenGL::EGL)
 target_include_directories(ab3d2-mock-openxr PRIVATE ${OPENXR_INCLUDE_DIRS})
 if(EXISTS "${CMAKE_SOURCE_DIR}/assets/native/manifest.json")
  add_test(NAME NativeOpenXR COMMAND ${CMAKE_COMMAND}
   -DMOCK=$<TARGET_FILE:ab3d2-mock-openxr> -DAPP=$<TARGET_FILE:alien-breed3d2-xr>
   -DASSETS=${CMAKE_SOURCE_DIR}/assets -DOUTPUT=${CMAKE_BINARY_DIR}/xr-fixture
   -P ${CMAKE_SOURCE_DIR}/native/tests/check-xr.cmake)
  set_tests_properties(NativeOpenXR PROPERTIES RUN_SERIAL TRUE)
 endif()
endif()
if(AB3D2_BUILD_GPU_TESTS AND EXISTS "${CMAKE_SOURCE_DIR}/assets/native/manifest.json")
 add_test(NAME NativeXRMenus COMMAND ${CMAKE_COMMAND}
  -DMOCK=$<TARGET_FILE:ab3d2-mock-openxr> -DAPP=$<TARGET_FILE:alien-breed3d2-xr>
  -DASSETS=${CMAKE_SOURCE_DIR}/assets -DOUTPUT=${CMAKE_BINARY_DIR}/xr-menu-fixture -DMODE=menu
  -P ${CMAKE_SOURCE_DIR}/native/tests/check-xr.cmake)
 set_tests_properties(NativeXRMenus PROPERTIES RUN_SERIAL TRUE)
endif()
