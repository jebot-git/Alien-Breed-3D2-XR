cmake_minimum_required(VERSION 3.20)
project(AlienBreed3D2XR VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS native/src/core/*.cpp)
add_library(ab3d2-original STATIC ${CORE_SOURCES})
target_include_directories(ab3d2-original PUBLIC native/include native/src/core)
target_compile_options(ab3d2-original PRIVATE -fwrapv -fno-strict-aliasing -Wno-narrowing)
add_executable(ab3d2-core-test native/tests/core-main.cpp)
target_link_libraries(ab3d2-core-test PRIVATE ab3d2-original)
enable_testing()
find_package(nlohmann_json REQUIRED)
target_link_libraries(ab3d2-core-test PRIVATE nlohmann_json::nlohmann_json)
add_executable(ab3d2-arithmetic-test native/tests/arithmetic.cpp)
target_link_libraries(ab3d2-arithmetic-test PRIVATE ab3d2-original)
add_test(NAME NativeArithmetic COMMAND ab3d2-arithmetic-test)
set(AB3D2_REFERENCE_FIXTURES "${CMAKE_SOURCE_DIR}/artifacts/native-oracle" CACHE PATH "Local original reference states; optional development fixtures")
foreach(level RANGE 0 15)
  if(EXISTS "${AB3D2_REFERENCE_FIXTURES}/level-${level}.ram" AND EXISTS "${AB3D2_REFERENCE_FIXTURES}/frames-${level}.json")
    add_test(NAME NativeOriginalLevel${level} COMMAND ab3d2-core-test "${AB3D2_REFERENCE_FIXTURES}/level-${level}.ram" "${AB3D2_REFERENCE_FIXTURES}/frames-${level}.json")
    set_tests_properties(NativeOriginalLevel${level} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  endif()
endforeach()
